-- Helper Function for Smooth UI Fading
local function FadeUIWindow(frame, show, duration, style)
    duration = duration or 0.3
    style = style or Enum.EasingStyle.Quad
    
    if show then
        frame.Visible = true
        frame.BackgroundTransparency = 1
        TweenService:Create(frame, TweenInfo.new(duration, style, Enum.EasingDirection.Out), {
            BackgroundTransparency = 0.1
        }):Play()
    else
        local tween = TweenService:Create(frame, TweenInfo.new(duration, style, Enum.EasingDirection.In), {
            BackgroundTransparency = 1
        })
        tween:Play()
        tween.Completed:Connect(function()
            frame.Visible = false
        end)
    end
end

        aiContentFrame.Visible = not toExec
        execContentFrame.Visible = toExec
        if toExec then
            execTabBtn.BackgroundColor3 = Color3.fromRGB(15, 125, 235)
            execTabBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            aiTabBtn.BackgroundColor3 = Color3.fromRGB(35, 35, 48)

    execTabBtn.MouseButton1Click:Connect(function() SwitchTab(true) end)

    -- GROQ API REQUEST HANDLER
    local function HandleGeneration()
        local query = promptInput.Text
        if query == "" then return end

        outputText.Text = "Generating custom script via Groq AI..."
        generateBtn.Text = "..."

        task.spawn(function()
            if not httpRequest then
                outputText.Text = "Error: Your executor does not support HTTP requests (request/http_request)."
                generateBtn.Text = "Send"
                return
            end

            local payload = HttpService:JSONEncode({
                model = "llama-3.3-70b-versatile",
                messages = {
                    {
                        role = "system",
                        content = "You are an expert Roblox Luau developer. Respond ONLY with valid, raw executable Luau code. Do NOT wrap the response in markdown blocks like ```lua or ```. Do NOT add conversational intro text."
                    },
                    {
                        role = "user",
                        content = "Write a complete Roblox client script for: " .. query
                    }
                },
                temperature = 0.2
            })

            local success, response = pcall(function()
                return httpRequest({
                    Url = GROQ_API_URL,
                    Method = "POST",
                    Headers = {
                        ["Content-Type"] = "application/json",
                        ["Authorization"] = "Bearer " .. GROQ_API_KEY
                    },
                    Body = payload
                })
            end)

            if success and response and response.Body then
                local decodeSuccess, decoded = pcall(function()
                    return HttpService:JSONDecode(response.Body)
                end)

                if decodeSuccess and decoded.choices and decoded.choices[1] and decoded.choices[1].message then
                    local code = decoded.choices[1].message.content
                    code = code:gsub("^```%w*\n?", ""):gsub("\n?```$", "") -- Strip markdown block tags if included
                    
                    scriptTextBox.Text = code
                    outputText.Text = "[Success]: Code generated and pasted to Executor Tab!"
                    generateBtn.Text = "Send"
                    promptInput.Text = ""
                    
                    SwitchTab(true)
                else
                    outputText.Text = "Error parsing AI response. Please check your API key."
                    generateBtn.Text = "Send"
                end
            else
                outputText.Text = "HTTP Error: Failed to communicate with Groq API."
                generateBtn.Text = "Send"
            end
        end)
    end
    
    generateBtn.MouseButton1Click:Connect(HandleGeneration)

        if code ~= "" and loadstring then
            local func, err = loadstring(code)
            if func then
                task.spawn(func)
                runBtn.Text = "EXECUTED!"
            else
                runBtn.Text = "ERROR!"
                warn("Execution error: " .. tostring(err))
            end
            task.wait(1.2)

    end)
